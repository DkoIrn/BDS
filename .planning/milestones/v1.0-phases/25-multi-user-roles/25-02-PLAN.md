**For server actions (src/lib/actions/*.ts):**

Replace this pattern:
```typescript
const { data: { user } } = await supabase.auth.getUser()
// ...
.eq('user_id', user.id)
```

With this pattern:
```typescript
import { requireOrgRole } from '@/lib/permissions'

const { data: { user } } = await supabase.auth.getUser()
if (!user) return { error: 'Not authenticated' }

const orgResult = await requireOrgRole(supabase, user.id, 'viewer') // or 'reviewer'/'admin' as needed
if ('error' in orgResult) return { error: orgResult.error }
const { orgId } = orgResult

// Replace .eq('user_id', user.id) with .eq('org_id', orgId) on projects
// For jobs/datasets/etc., RLS handles it -- just remove the explicit user_id filter
```
