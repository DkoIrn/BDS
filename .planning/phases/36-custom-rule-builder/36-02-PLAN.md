  <name>Task 1: TypeScript types and constants for custom rules</name>
  <files>src/lib/types/custom-rules.ts</files>
  <action>
    Create `src/lib/types/custom-rules.ts` with types that mirror the backend Pydantic models:

    ```typescript
    // Rule type discriminator
    export type RuleType = "threshold" | "comparison" | "null_check"

    // Single condition
    export interface Condition {
      id: string           // client-side temp ID via crypto.randomUUID()
      column: string       // mapped column type (e.g., "dob", "doc")
      ruleType: RuleType
      operator: string     // e.g., ">", "<", "is_null"
      value?: number | string    // for threshold
      compareColumn?: string     // for comparison
    }

    // Group of conditions with AND/OR logic
    export interface ConditionGroup {
      id: string
      logic: "AND" | "OR"
      conditions: Condition[]
      groups: ConditionGroup[]   // nested groups, max depth 2
    }

    // Full rule definition (what gets saved)
    export interface CustomRuleDefinition {
      name: string
      description: string
      severity: "critical" | "warning" | "info"
      rootGroup: ConditionGroup
    }

    // Database record (returned from API)
    export interface CustomRule {
      id: string
      profile_id: string
      user_id: string
      org_id: string
      name: string
      description: string
      severity: "critical" | "warning" | "info"
      rule_definition: ConditionGroup   // rootGroup stored as JSON
      enabled: boolean
      created_at: string
      updated_at: string
    }

    // Test result from backend
    export interface RuleTestResult {
      matching_rows: number
      total_rows: number
      sample_matches: Array<{ row_number: number; values: Record<string, unknown> }>
      truncated?: boolean
    }
    ```

    Also export operator constant objects:

    ```typescript
    export const THRESHOLD_OPERATORS = [
      { value: ">", label: "greater than" },
      { value: "<", label: "less than" },
      { value: ">=", label: "greater than or equal" },
      { value: "<=", label: "less than or equal" },
      { value: "==", label: "equals" },
      { value: "!=", label: "not equal to" },
    ] as const

    export const COMPARISON_OPERATORS = THRESHOLD_OPERATORS  // same set

    export const NULL_OPERATORS = [
      { value: "is_null", label: "is empty" },
      { value: "is_not_null", label: "is not empty" },
    ] as const
    ```

    Export helper functions:
    - `createEmptyCondition()`: returns a Condition with crypto.randomUUID() id, empty column, threshold type, ">" operator
    - `createEmptyGroup(logic?: "AND" | "OR")`: returns a ConditionGroup with one empty condition
    - `getOperatorsForType(ruleType: RuleType)`: returns the appropriate operator array
    - `getNestingDepth(group: ConditionGroup, current?: number)`: recursively counts depth
    - `canAddNestedGroup(group: ConditionGroup)`: returns true if depth < 2
  </action>
